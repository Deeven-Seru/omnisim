Install Once. Protect Every Pull Request.

Run pre-flight invariant audits locally via CLI or block breaking infrastructure changes in GitHub Actions.

name: OmniSim Pre-Flight Invariant Gatekeeper
on:
  pull_request:
    paths:
      - 'terraform/**'
      - 'k8s/**'

jobs:
  simulate:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Pre-Flight Invariant Simulation
        uses: Deeven-Seru/omnisim@v1
        with:
          terraform-dir: 'terraform'
          fail-on-breach: 'true'
